% ---- 경로 -------------------------------------------------------------------

% ===================== 오른쪽 : 무엇을 사고 무엇을 파는가 ===================
\begin{scope}[shift={(96,0)}]
  \node[font=\small\bfseries, anchor=north west] at (0,28) {the trade};
  \node[anchor=north west, align=left, text width=76mm, font=\scriptsize] at (0,23) {%
    One and the same $y_e$. Only $\Delta$ has changed, and the command has changed
    by a factor of three.\\[5pt]
    \textcolor{warn}{\textbf{$\Delta$ too small}} — the arctan saturates almost at
    once, the vessel turns nearly perpendicular to the path and arrives with speed
    across it. Measured: $0.939$\,m of overshoot at $\Delta = 2$\,m.\\[4pt]
    \textcolor{accent}{\textbf{$\Delta$ too large}} — the correction is gentle and
    the vessel closes slowly. At $\Delta = 30$\,m it never gets within $0.4$\,m
    before the $60$\,m leg ends, and is still $1.29$\,m out.\\[5pt]
    The rule of thumb is $\Delta = 2$ to $5$ hull lengths; for the $2$\,m Otter
    that is $4$ to $10$\,m, and this course uses $8$\,m. \textbf{Section E is the
    evidence, and the rule of thumb only the sanity check.}\\[5pt]
    A third cost appears in §4-7: the steady offset a current leaves is
    $\Delta\tan\beta_c$, \textbf{linear in $\Delta$}.};
\end{scope}
