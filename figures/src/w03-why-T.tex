% =====================================================================
% 노트의 세 줄 / the three lines of the note
% =====================================================================
\node[anchor=north west, align=left, font=\small] at (3.7,2.95)
  {\textbf{1.\ \ read the three forces off the picture}\\[5pt]
   $\begin{bmatrix} X \\ Y \\ N \end{bmatrix}
    = \begin{bmatrix} F_L + F_R \\[2pt] 0 \\[2pt] b\,(F_L - F_R) \end{bmatrix}$};

%  설명은 행렬 **오른쪽 바깥**에. 가까이 두면 행렬 위에 찍힌다 (렌더해서 고침)
\node[note, anchor=north west, align=left] at (7.65,1.95)
  {both push forward, so they \textbf{add}\\[11pt]
   neither pushes sideways:\\ \textbf{this row is empty, always}\\[9pt]
   opposite arms, so they \textbf{subtract}};

\node[anchor=north west, align=left, font=\small] at (3.7,-0.55)
  {\textbf{2.\ \ drop the empty row --- it is square now}\\[5pt]
   $\begin{bmatrix} X \\ N \end{bmatrix}
    = \underbrace{\begin{bmatrix} 1 & 1 \\ b & -b \end{bmatrix}}_{\textstyle \mathbf{T}}
      \begin{bmatrix} F_L \\ F_R \end{bmatrix}$};

\node[note, anchor=north west, align=left] at (7.65,-1.05)
  {$\mathbf{T}$ is \textbf{geometry}, nothing else:\\
   it never changes while the\\
   propellers stay bolted where they are};

\node[anchor=north west, align=left, font=\small] at (3.7,-3.05)
  {\textbf{3.\ \ invert it --- that is the allocator}\\[5pt]
   $\begin{bmatrix} F_L \\ F_R \end{bmatrix}
    = \mathbf{T}^{-1}\begin{bmatrix} X \\ N \end{bmatrix}
    = \begin{bmatrix} \tfrac12 & \tfrac{1}{2b} \\[3pt] \tfrac12 & -\tfrac{1}{2b} \end{bmatrix}
      \begin{bmatrix} X \\ N \end{bmatrix}$};

\node[note, anchor=north, align=center] at (4.95,-5.15) {known};
\draw[lead] (4.95,-5.05) -- (4.95,-4.70);

\node[note, anchor=north, align=center] at (8.15,-5.15)
  {what the two\\ controllers asked for};
\draw[lead] (8.15,-5.05) -- (8.15,-4.70);

  {$\det \mathbf{T} = -2b \neq 0$, so the inverse exists and the split is unique:

   and the moment is added to one and taken from the other};
