% ===== (a) 부호는 반평면을 읽는 것이고, 접기는 점대칭이다 ==================
\begin{scope}[shift={(0,0)}]
  \fill[rev] (-2.40,-2.40) rectangle (0,2.40);
  \fill[fwd] (0,0) -- (1.626,1.626) -- (1.626,-1.626) -- cycle;

  \draw[ax] (-2.50,0) -- (2.60,0);
  \draw[ax] (0,-2.50) -- (0,2.50);
  \node[lab, anchor=west]       at (2.62,0)      {$f_{x,i}$};
  \node[lab, anchor=south east] at (-0.06,2.42)  {$f_{y,i}$};

  \draw[wall] (0,0) -- ( 1.626, 1.626);
  \draw[wall] (0,0) -- ( 1.626,-1.626);
  \node[lab, anchor=south west, text=accent] at (1.72, 1.62) {$+\alpha_{\max}$};
  \node[lab, anchor=north west, text=accent] at (1.72,-1.62) {$-\alpha_{\max}$};

  % 요구된 성분 쌍: (-43.30, +25.00) N  →  150 도
  \draw[fvec] (0,0) -- (-1.732,1.000);
  \node[lab, anchor=south east] at (-1.78,1.06)
    {asked\\$(-43.30,\;+25.00)$ N};
  \draw[arc] (0.55,0) arc (0:150:0.55);
  \node[lab, anchor=east] at (-0.34,0.62) {$150^\circ$};

  % 점대칭 — 같은 길이, 정확히 pi
  \draw[mirror] (-1.732,1.000) -- (1.800,-1.039);
  \draw[gvec]   (0,0) -- (1.732,-1.000);
  \node[lab, anchor=west, text=knownc] at (1.86,-0.92)
    {$T_i=-50$ N\\$\delta_i=-30^\circ$};

  \node[lab, text=warn]   at (-1.50,-2.30) {$\sigma_i=-1$\\astern half};
  \node[lab, text=accent] at ( 1.02,-2.30) {$\sigma_i=+1$\\ahead half};

  \node[tlab] at (0.05,3.05)
    {(a)~$\sigma_i$ reads \textbf{which half-plane} the demand lies in};

  \node[blk] at (0.05,-2.95)
    {The demand points into the astern half, so $\sigma_i=-1$. Multiplying
     \emph{both} \texttt{atan2} arguments by $\sigma_i$ reflects the arrow
     through the origin: same length, direction turned through exactly $\pi$,
     and the price of that turn is $T_i<0$.};
\end{scope}

% ===== (b) 접지 않으면 벽이 요구를 뒤집는다 ================================
\begin{scope}[shift={(8.6,0)}]
  \fill[fwd] (0,0) -- (1.626,1.626) -- (1.626,-1.626) -- cycle;

  \draw[ax] (-2.50,0) -- (2.60,0);
  \draw[ax] (0,-2.50) -- (0,2.50);
  \node[lab, anchor=west]       at (2.62,0)      {$f_{x,i}$};
  \node[lab, anchor=south east] at (-0.06,2.42)  {$f_{y,i}$};

  \draw[wall] (0,0) -- ( 1.626, 1.626);
  \draw[wall] (0,0) -- ( 1.626,-1.626);

  % 요구: 똑바로 뒤로 50 N
  \draw[fvec] (0,0) -- (-2.000,0);
  \node[lab, anchor=south] at (-1.00,0.12) {asked $(-50,\;0)$ N};

  % 접지 않은 결과: 180 도를 45 도 벽으로 자른다
  \draw[bvec] (0,0) -- (1.414,1.414);
  \node[lab, anchor=south west, text=warn] at (1.46,1.50)
    {no fold: the $180^\circ$ command\\
     is clipped to the $+45^\circ$ wall\\
     and delivers $(+35.36,\;+35.36)$ N};

  % 접은 결과: 벽이 아무것도 하지 않는다
  \draw[gvec] (0,-0.32) -- (-2.000,-0.32);
  \node[lab, anchor=north east, text=knownc] at (0.10,-0.48)
    {folded: $T_i=-50$ N at $\delta_i=0$,\\exactly the demand};

  \node[tlab] at (0.05,3.05)
    {(b)~without the fold the servo wall \textbf{reverses} the demand};

  \node[blk] at (0.05,-2.95)
    {Straight astern is $180^\circ$, the farthest point from the wedge.
     Clipping that angle keeps $T_i>0$, so the unit pushes the hull
     \emph{forward} and adds a sway that was never asked for. Folding first
     leaves the demand untouched.};
\end{scope}

% ===== 아래 한 줄 ==========================================================
\node[note] at (4.32,-4.85)
  {A propeller running astern is $T_i<0$ at a small angle, never $T_i>0$ at

   $\sigma_i f_{x,i}\ge 0$ holds by construction,\\
   which is what makes the servo limit the pair of straight lines
   $\lvert f_{y,i}\rvert \le f_{x,i}\tan\alpha_{\max}$ used in \S7-10.};
