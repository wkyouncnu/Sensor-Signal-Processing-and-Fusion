% ===================== (b) LOS =============================================
\begin{scope}[shift={(76,0)}]
  \node[font=\small\bfseries, anchor=west] at (0,22) {(b)\quad LOS — aim $\Delta$ ahead on the path};
  \node[note, anchor=west] at (0,17.5) {$\psi_d = \pi_p - \arctan(y_e/\Delta)$,\ \ $\Delta = 8$\,m};

  \draw[pathc, line width=0.7pt, densely dashed] (0,0) -- (\XE,0);
  \filldraw[pathc, fill=white, line width=0.6pt] (0,0)   circle (0.9);
  \filldraw[pathc, fill=white, line width=0.6pt] (\XE,0) circle (0.9);

  \draw[accent, line width=0.9pt] plot coordinates{\LOStrack};
  \node[accent, font=\scriptsize, anchor=north west] at (6,-8) {the track};

  \draw[errc, line width=0.7pt] (24,0) -- (24,-0.07);
  \node[errc, font=\scriptsize, anchor=west] at (25,-3.4) {$y_e = 0.11$\,m here};

  \node[anchor=north west, align=left, text width=62mm, font=\scriptsize] at (0,-13) {%
    The law regulates the distance to the \textbf{line}. The aim point moves along
    the path as the vessel does, so the correction fades to zero exactly when the
    vessel reaches the path, and the rest of the leg is run along it.};
\end{scope}
