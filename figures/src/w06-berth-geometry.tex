%       뒷받침하지 않는 기호를 강의에 싣지 않는다 (§15-바로 아래 교훈).
%    ② 넓은 산문 칸 — `.md` 로 옮겼다.
%
%  화면 = NED 를 위에서 본 것.  screen x = east,  screen y = north.
%  \NE{북}{동} 이 **참 NED 수치**를 받아 접안지점을 원점으로 옮긴다. 그래서
%  이 파일의 기하는 W06_vars.m 의 숫자를 그대로 읽는다.
%
%  축척 1 m = 5 mm. 선체는 **실척**이다 — Otter LOA 2.0 m = 10 mm.
%  캡스톤은 고정 12.8 mm 글리프를 썼고, 그러면 R_berth = 0.40 m 가 선체보다
%  얼마나 작은지가 거짓말이 된다.
%
%  주석만 한국어다. 조판되는 글자에 한글을 넣으면 pdflatex 이 멈춘다.

\documentclass[border=6pt]{standalone}
\input{gnc-style}

\definecolor{bodyc}{HTML}{1D4ED8}

\pgfmathsetmacro{\PS}{5}                                   % 1 m = 5 mm

%  참 NED 수치 -> 화면 좌표 / true NED numbers -> screen coordinates
\newcommand{\NE}[2]{({((#2) - 1.2)*\PS},{((#1) - 40)*\PS})}

%  선체 하나, 실척 / one hull, to scale.  #1 north  #2 east  #3 psi [deg]  #4 colour
%  shift= 에는 좌표 매크로를 줄 수 없다 — 숫자 식으로 푼다
%
%  꼭짓점은 **다섯**이다 — 좌현 선미, 좌현 어깨, 뱃머리, 우현 어깨, 우현 선미.
%  넷만 적고 cycle 하면 우현 어깨에서 좌현 선미로 대각선이 그어져 선체가
%  **연꼴**이 된다 (교훈 lxxxii).
\newcommand{\hull}[4]{%
  \begin{scope}[shift={({((#2) - 1.2)*\PS},{((#1) - 40)*\PS})}, rotate={90 - (#3)}]
    \fill[#4!12]                  (-5.0,-2.7) -- (2.75,-2.7) -- (5.0,0) -- (2.75,2.7) -- (-5.0,2.7) -- cycle;
    \draw[#4, line width=0.6pt]   (-5.0,-2.7) -- (2.75,-2.7) -- (5.0,0) -- (2.75,2.7) -- (-5.0,2.7) -- cycle;
    \draw[#4, line width=0.3pt]   (-5.0,0) -- (5.0,0);
  \end{scope}}

\tikzset{
  io/.style    = {font=\footnotesize, ink, inner sep=1.6pt},

  enote/.style = {font=\scriptsize, errc, inner sep=1.4pt},
  tag/.style   = {draw=soft, line width=0.5pt, fill=black!3, rounded corners=1.5pt,
                  font=\scriptsize, inner sep=2.2pt, align=left},
  dimn/.style  = {ink, line width=0.5pt,
                  {Stealth[length=3.4pt,width=2.6pt]}-{Stealth[length=3.4pt,width=2.6pt]}},
  ext/.style   = {soft, line width=0.35pt, densely dotted},
}
